import { useState, useEffect } from "react";
import Home from "./Home";
import "./LandingPage.css";

function LandingPage() {

  const [showButton, setShowButton] = useState(false);
  const [enterHome, setEnterHome] = useState(false);

  // ⌨️ Typewriter phrases
  const phrases = [
    "Learn Together...",
    "Teach Others...",
    "Grow Together...",
    "Connect Globally...",
    "Exchange Knowledge..."
  ];

  const [index, setIndex] = useState(0);

  // ⌨️ Auto changing text
  useEffect(() => {
    const interval = setInterval(() => {
      setIndex((prev) => (prev + 1) % phrases.length);
    }, 2000);

    return () => clearInterval(interval);
  }, []);

  // ⏳ Show Explore button after 3s
  useEffect(() => {
    const timer = setTimeout(() => {
      setShowButton(true);
    }, 3000);

    return () => clearTimeout(timer);
  }, []);

  // 🚀 Navigate to Home
  if (enterHome) {
    return (
      <div className="fade-home">
        <Home />
      </div>
    );
  }

  return (
    <div className="landing">

      {/* 🎈 Floating bubbles */}
      <div className="bubble react">⚛ React</div>
      <div className="bubble ai">🤖 AI</div>
      <div className="bubble python">🐍 Python</div>
      <div className="bubble java">☕ Java</div>
      <div className="bubble guitar">🎸 Guitar</div>
      <div className="bubble photo">📸 Photography</div>

      {/* 🧠 CENTER CONTENT */}
      <div className="center-box">

        <h1 className="title">EduConnect</h1>

        <h2 className="subtitle">
          Learn. Teach. Grow Together.
        </h2>

        {/* ⌨️ TYPEWRITER TEXT */}
        <p className="typewriter-text">
          {phrases[index]}
        </p>

        {/* ✨ Explore Button */}
        {showButton && (
          <button
            className="start-btn"
            onClick={() => setEnterHome(true)}
          >
            ✨ Explore
          </button>
        )}

      </div>

    </div>
  );
}

export default LandingPage;
